💻 Python Game Code
import random

secret = random.randint(1, 100)
attempts = 0
while True:
    guess = int(input("Enter guess: "))
    attempts += 1
    if guess < secret:
        print("Too Low!")
    elif guess > secret:
        print("Too High!")
    else:
        print(f"Correct in {attempts} attempts!")
        break
