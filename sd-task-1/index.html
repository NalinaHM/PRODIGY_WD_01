// C++ Core Conversion Logic
#include <iostream>
using namespace std;

int main() {
    double celsius = 25.0;
    double fahrenheit = (celsius * 9.0 / 5.0) + 32.0;
    double kelvin = celsius + 273.15;
    cout << "Fahrenheit: " << fahrenheit << " F\n";
    cout << "Kelvin: " << kelvin << " K\n";
    return 0;
}
