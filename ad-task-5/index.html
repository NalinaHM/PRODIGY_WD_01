⚙️ Retrofit REST Client & Coroutines

Leverages Retrofit 2 for async HTTP requests with Kotlin Coroutines lifecycleScope.launch for thread-safe UI rendering.

📄 WeatherApiService.kt & MainActivity.kt Snippet

interface WeatherApiService {
    @GET("weather")
    suspend fun getCurrentWeather(
        @Query("q") city: String,
        @Query("appid") apiKey: String,
        @Query("units") units: String = "metric"
    ): Response
}

class MainActivity : AppCompatActivity() {
    private fun fetchWeather(city: String) {
        lifecycleScope.launch {
            val response = RetrofitClient.apiService.getCurrentWeather(city, API_KEY)
            if (response.isSuccessful) {
                binding.tempText.text = "${response.body()?.main?.temp}°C"
            }
        }
    }
}
